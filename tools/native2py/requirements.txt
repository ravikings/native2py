# Installs native2py itself (editable) plus everything needed to build,
# test, and document the generated services. Equivalent to:
#   pip install -e ".[docs,build,test]"
# Prefer that form if you only need a subset — see pyproject.toml for the
# [build]/[test]/[docs] groups broken out individually.
#
# REPRODUCIBILITY: the ranges below are the DECLARED requirements. They are
# deliberately not exact pins — pyproject.toml is a library manifest and
# over-pinning it propagates conflicts to consumers. For a reproducible,
# hash-verified development or CI environment install through the lock file
# instead:
#
#   pip install --require-hashes -r constraints.txt
#   pip install --no-deps -e .
#
# constraints.txt pins every direct AND transitive package to an exact version
# with SHA-256 hashes; see its header for how to regenerate it.

-e .

# --- native2py CLI runtime (also in pyproject.toml [project.dependencies]) ---
click>=8.1
pyyaml>=6.0

# --- C++ AST parser (pyproject.toml [clang]) ---
# Ships its own libclang shared library. Without it the C++ front end falls
# back to the regex reader: no preprocessor, no templates, no typedef
# resolution. Set NATIVE2PY_LIBCLANG to use a system libclang instead.
libclang>=16

# --- Fortran parse-tree backend (pyproject.toml [fparser]) ---
# Pure Python, so it adds no toolchain. Without it the Fortran front end uses
# the regex reader and every test in tests/test_fortran_fparser.py skips —
# which is why it is installed here and in CI rather than left to chance.
fparser>=0.2,<0.3

meson>=1.1
ninja>=1.11

# --- testing generated services (pyproject.toml [test]) ---

# No longer present by default in a Python 3.12 venv; tests/test_golden.py
